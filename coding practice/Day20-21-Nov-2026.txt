
# instance type of EC2 and EKS should be same
# to see all clusters
aws eks list-clusters --region ap-southeast-2

# to see all subnet Ids
aws eks describe-cluster \
  --name Bodyprime_Nutrition_EKS \
  --region ap-southeast-2 \
  --query 'cluster.resourcesVpcConfig.subnetIds' \
  --output table

# to see all role arn
aws iam get-role \
  --role-name BodyprimeEKSNodeRole \
  --query 'Role.Arn' \
  --output text

# to create nodeGroups
aws eks create-nodegroup \
  --cluster-name Bodyprime_Nutrition_EKS \
  --nodegroup-name second-worker-group \
  --node-role arn:aws:iam::877133239596:role/BodyprimeEKSNodeRole \
  --subnets subnet-07e4740d088ab10a6 subnet-0a895875fe805c2be subnet-0439bb06f186fe83b \
  --instance-types t3.micro \
  --capacity-type ON_DEMAND \
  --scaling-config minSize=1,maxSize=5,desiredSize=1 \
  --region ap-southeast-2

# to see the nodeGroups status
aws eks describe-nodegroup \
  --cluster-name Bodyprime_Nutrition_EKS \
  --nodegroup-name second-worker-group \
  --region ap-southeast-2 \
  --query 'nodegroup.status' \
  --output text

# to see the nodeGroups
aws eks list-nodegroups \
  --cluster-name Bodyprime_Nutrition_EKS \
  --region ap-southeast-2

# to see instance type and scaling configurations
aws eks describe-nodegroup \
  --cluster-name Bodyprime_Nutrition_EKS \
  --nodegroup-name second-worker-group \
  --region ap-southeast-2 \
  --query 'nodegroup.{InstanceTypes:instanceTypes,Scaling:scalingConfig,Status:status}'

# to see scaling configurations
aws eks describe-nodegroup \
  --cluster-name Bodyprime_Nutrition_EKS \
  --nodegroup-name second-worker-group \
  --region ap-southeast-2 \
  --query 'nodegroup.scalingConfig'

# to update scaling configurations
aws eks update-nodegroup-config \
  --cluster-name Bodyprime_Nutrition_EKS \
  --nodegroup-name second-worker-group \
  --scaling-config minSize=1,maxSize=2,desiredSize=2 \
  --region ap-southeast-2

















- name: backup and restore files
  hosts: all
  become: yes
  tasks:
    - name: create backup of /etc directory
      command: tar -czvf /tmp/etc-backup.tar.gz /etc
    - name: copy backup to remote server
      copy:
        src: /tmp/etc-backup.tar.gz
        dest: /tmp/etc-backup.tar.gz
    - name: restore backup from remote server
      command: tar -xzvf /tmp/etc-backup.tar.gz -C /

pipeline {
    agent any
    stages {
        stage('Backup') {
            steps {
                echo 'Backing up Jenkins workspace...'
                sh 'tar -czvf /tmp/jenkins-backup.tar.gz ${WORKSPACE}'
            }
        }
        stage('Restore') {
            steps {
                echo 'Restoring Jenkins workspace...'
                sh 'tar -xzvf /tmp/jenkins-backup.tar.gz -C ${WORKSPACE}'
            }
        }
    }
}
