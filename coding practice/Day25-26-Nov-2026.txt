- name: implement SSH key authentication
  hosts: all
  become: yes
  tasks:
    - name: generate SSH key pair
      command: ssh-keygen -t rsa -b 2048 -f /home/user/.ssh/id_rsa -N ""
    - name: Add SSH public key to authorized_keys
      authorized_key:
        user: user
        state: present
        key: "{{ lookup('file', '/home/user/.ssh/id_rsa.pub') }}"

apiVersion: networking.k8s.io/v1
kind: NetworkPolicy
metadata:
  name: gourav-web-policy
spec:
  podSelector:
    matchLabels:
      app: gourav-web-app
  policyTypes:
    - Ingress
    - Egress
  ingress:
    - from:
        - podSelector:
            matchLabels:
              app: gourav-api
  egress:
    - to:
        - podSelector:
            matchLabels:
              app: gourav-database







pipeline {
    agent any
    environment {
        AWS_ACCESS_KEY_ID = credentials('aws-access-key')
        AWS_SECRET_ACCESS_KEY = credentials('aws-secret-key')
    }
    stages {
        stage('Build') {
            steps {
                echo 'building Docker image...'
                docker.build('gourav-web-app')
            }
        }
        stage('Deploy') {
            steps {
                echo 'deploying to AWS EC2...'
                sh 'aws ec2 run-instances --image-id ami-0wedasi9abcd1234 --count 1 --instance-type t2.micro'
            }
        }
    }
}








stages:
  - build
  - deploy

build:
  stage: build
  script:
    - docker build -t gourav-web-app .

deploy:
  stage: deploy
  script:
    - echo "Deploying with secure credentials"
    - aws ec2 run-instances --image-id ami-0wedasi9abcd1234 --count 1 --instance-type t2.micro --access-key $AWS_ACCESS_KEY --secret-key $AWS_SECRET_KEY
