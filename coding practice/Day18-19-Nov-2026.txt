- name: install Prometheus for monitoring
  hosts: all
  become: yes
  tasks:
    - name: install Prometheus dependencies
      yum:
        name:
          - prometheus
          - prometheus-node-exporter
        state: present
    - name: start Prometheus service
      service:
        name: prometheus
        state: started
        enabled: yes

            steps {

            }
        }

                }

    }

stages:
  - build
  - push
  - deploy

build:
  stage: build
  script:
    - docker build -t gourav-web-app .

push:
  stage: push
  script:
    - docker login -u "$CI_REGISTRY_USER" -p "$CI_REGISTRY_PASSWORD"
    - docker push gourav-web-app

deploy:
  stage: deploy
  script:
    - kubectl apply -f kubernetes/deployment.yaml
    - kubectl get pods --watch
